# ==============================================================================
# [IMMUTABLE CONFIGURATION - MANUAL CONTROL ONLY]
# ------------------------------------------------------------------------------
# DO NOT ALLOW AI AGENTS OR AUTOMATION TO MODIFY THE PIPELINE LOGIC.
# Current CI State: 
# 1. Automatic change detection: DISABLED.
# 2. Previous failed/skipped job recovery: DISABLED.
# 3. Execution: Force full run on every trigger.
# Any modification to this file requires manual human intervention and 
# signature validation to prevent regression or instability.
# ==============================================================================

name: 🛠️ SupremeAI Nightly Maintenance

on:
  schedule:
    - cron: '0 2 * * 0'       # Supabase Backup & Cloud Cleanup (Sundays 2 AM UTC)
    - cron: '0 0 * * *'       # Prompt Evaluation & Auto-Coverage (Midnight UTC)
  workflow_dispatch:
    inputs:
      job_to_run:
        description: 'Select maintenance job to run'
        required: true
        type: choice
        options:
          - all
          - db-backup
          - ai-evaluation
          - cleanup
          - dependency-update

env:
  PYTHON_VERSION: '3.11'
  NODE_VERSION: '24'

jobs:
  # ----------------------------------------------------
  # 🗄️ DATABASE BACKUP TO CLOUDFLARE R2
  # ----------------------------------------------------
  db-backup:
    name: 🗄️ Supabase Backup to R2
    if: github.event_name == 'schedule' || github.event.inputs.job_to_run == 'all' || github.event.inputs.job_to_run == 'db-backup'
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - name: Install PostgreSQL Client
        run: sudo apt-get update && sudo apt-get install -y postgresql-client
      - name: Dump & Upload to R2
        env:
          SUPABASE_DB_URL: ${{ secrets.SUPABASE_DATABASE_URL }}
          AWS_ACCESS_KEY_ID: ${{ secrets.R2_ACCESS_KEY }}
          AWS_SECRET_ACCESS_KEY: ${{ secrets.R2_SECRET_KEY }}
          R2_ACCOUNT_ID: ${{ secrets.R2_ACCOUNT_ID }}
          BUCKET_NAME: ${{ secrets.R2_BUCKET_NAME }}
        run: |
          FILENAME="supremeai_db_backup_$(date +%Y%m%d_%H%M%S).dump"
          pg_dump "$SUPABASE_DB_URL" -F c -f $FILENAME
          aws s3 cp $FILENAME s3://${BUCKET_NAME}/database_backups/ --endpoint-url "https://${R2_ACCOUNT_ID}.r2.cloudflarestorage.com" --region auto

  # ----------------------------------------------------
  # 🤖 AI PROMPT EVALUATION & COVERAGE
  # ----------------------------------------------------
  ai-evaluation:
    name: 🧪 AI Evaluation & Tests
    if: github.event_name == 'schedule' || github.event.inputs.job_to_run == 'all' || github.event.inputs.job_to_run == 'ai-evaluation'
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-python@v5
        with:
          python-version: ${{ env.PYTHON_VERSION }}
          cache: 'pip'
      - name: Install Dependencies
        run: |
          pip install poetry
          cd backend && poetry install --with dev --without ml
      - name: Auto-Generate Missing Tests
        env:
          GEMINI_API_KEY: ${{ secrets.GEMINI_API_KEY }}
        run: |
          cd backend
          poetry run pytest --cov=core --cov-report=xml || true
          poetry run python -m tools.auto_test_generator --report-path coverage.xml --min-coverage 75

  # ----------------------------------------------------
  # 🧹 CLOUD & CACHE CLEANUP
  # ----------------------------------------------------
  cleanup:
    name: 🧹 Cloud Run & Cache Cleanup
    if: github.event_name == 'schedule' || github.event.inputs.job_to_run == 'all' || github.event.inputs.job_to_run == 'cleanup'
    runs-on: ubuntu-latest
    permissions:
      contents: read
      id-token: write
    steps:
      - uses: actions/checkout@v4
      - uses: google-github-actions/auth@v2
        with:
          credentials_json: ${{ secrets.GCP_SA_KEY }}
      - uses: google-github-actions/setup-gcloud@v2
      - name: Delete old Cloud Run revisions
        run: |
          gcloud run revisions list --service=supremeai-api --region=${{ vars.GCP_REGION || 'us-central1' }} --filter="traffic.percent=0" --format="value(name)" --sort-by="~createTime" | tail -n +6 | xargs -r -I {} gcloud run revisions delete {} --region=${{ vars.GCP_REGION || 'us-central1' }} --quiet
      - name: Install Redis client
        run: python -m pip install redis
      - name: Run Cache Cleanup
        env:
          REDIS_URL: ${{ secrets.REDIS_URL }}
        run: |
          python scripts/cache_cleanup.py

  # ----------------------------------------------------
  # 🧹 GITHUB ACTIONS CACHE PRUNING
  # ----------------------------------------------------
  cache-prune:
    name: 🧹 Prune GitHub Caches
    if: github.event_name == 'schedule' || github.event.inputs.job_to_run == 'all' || github.event.inputs.job_to_run == 'cleanup'
    runs-on: ubuntu-latest
    permissions:
      actions: write
      contents: read
    steps:
      - name: Checkout code
        uses: actions/checkout@v4

      - name: Force Purge Unnecessary Old Caches (Smartest Clean)
        env:
          GH_TOKEN: ${{ secrets.GITHUB_TOKEN }}
        run: |
          echo "🧹 Initiating Force Cache Purge Engine..."
          gh cache list --repo ${{ github.repository }} --limit 100 | awk '{print $1}' | while read -r cache_key; do
            if [ ! -z "$cache_key" ]; then
              echo "🗑️ Force deleting stale cache key -> $cache_key"
              gh cache delete "$cache_key" --repo ${{ github.repository }} || true
            fi
          done
          echo "✅ Caches optimized successfully! Runtime minutes saved."

  # ----------------------------------------------------
  # 📦 DEPENDENCY UPDATE (Dependabot বিকল্প)
  # ----------------------------------------------------
  # বাংলা মন্তব্য: Dependabot-এর বদলে নিজস্ব স্কেডুল্ড জব ব্যবহার করা হচ্ছে
  # কারণ Dependabot প্রতিটি আপডেটের জন্য আলাদা ব্রাঞ্চ/PR তৈরি করে
  # রিপোজিটরির হিস্ট্রি অগোছালো করে ফেলত।
  # এই জব সব ইকোসিস্টেমের আপডেট একত্রিত করে একটি মাত্র PR তৈরি করে।
  dependency-update:
    name: 📦 Consolidated Dependency Update
    if: github.event_name == 'schedule' || github.event.inputs.job_to_run == 'all' || github.event.inputs.job_to_run == 'dependency-update'
    runs-on: ubuntu-latest
    permissions:
      contents: write
      pull-requests: write
    steps:
      - uses: actions/checkout@v4
        with:
          fetch-depth: 0
          ref: ${{ github.event.repository.default_branch }}

      - name: Configure Git
        run: |
          git config user.name "SupremeAI Bot"
          git config user.email "bot@supremeai.dev"

      # ------------------------------------------------
      # 🟢 Node.js / pnpm — ফ্রন্টেন্ড ডিপেন্ডেন্সি আপডেট
      # ------------------------------------------------
      - uses: actions/setup-node@v4
        with:
          node-version: ${{ env.NODE_VERSION }}

      - uses: pnpm/action-setup@v4

      - name: "📦 [npm] Update root & frontend dependencies"
        run: |
          echo "🔍 pnpm দিয়ে ডিপেন্ডেন্সি আপডেট চেক করা হচ্ছে..."
          pnpm update --no-frozen-lockfile 2>/dev/null || true
          echo "✅ Node.js ডিপেন্ডেন্সি আপডেট সম্পন্ন"

      # ------------------------------------------------
      # 🐍 Python / Poetry — ব্যাকএন্ড ডিপেন্ডেন্সি আপডেট
      # ------------------------------------------------
      - uses: actions/setup-python@v5
        with:
          python-version: ${{ env.PYTHON_VERSION }}
          cache: 'pip'

      - name: "📦 [pip] Update backend dependencies"
        run: |
          echo "🔍 Poetry দিয়ে ব্যাকএন্ড ডিপেন্ডেন্সি আপডেট চেক করা হচ্ছে..."
          pip install poetry
          cd backend
          poetry update --no-interaction 2>/dev/null || true
          cd ..
          echo "✅ Python ডিপেন্ডেন্সি আপডেট সম্পন্ন"

      # ------------------------------------------------
      # 🐦 Flutter / Dart — মোবাইল ডিপেন্ডেন্সি আপডেট
      # ------------------------------------------------
      - uses: subosito/flutter-action@v2
        with:
          channel: 'stable'

      - name: "📦 [pub] Update mobile dependencies"
        run: |
          echo "🔍 Flutter দিয়ে মোবাইল ডিপেন্ডেন্সি আপডেট চেক করা হচ্ছে..."
          cd apps/mobile
          flutter pub upgrade 2>/dev/null || true
          cd ../..
          echo "✅ Flutter ডিপেন্ডেন্সি আপডেট সম্পন্ন"

      # ------------------------------------------------
      # 📊 রিপোর্ট তৈরি ও একত্রিত PR
      # ------------------------------------------------
      - name: "📊 Generate update summary & create PR"
        env:
          GH_TOKEN: ${{ secrets.GITHUB_TOKEN }}
        run: |
          # বাংলা মন্তব্য: পরিবর্তন আছে কিনা চেক করা হচ্ছে
          if git diff --quiet && git diff --cached --quiet; then
            echo "✅ কোনো ডিপেন্ডেন্সি আপডেট পাওয়া যায়নি। সব আপ-টু-ডেট!"
            exit 0
          fi

          echo "📦 ডিপেন্ডেন্সি পরিবর্তন পাওয়া গেছে — PR তৈরি করা হচ্ছে..."

          # ব্রাঞ্চের নাম — তারিখ সহ যাতে প্রতিদিন ইউনিক হয়
          DATE_TAG=$(date +%Y%m%d)
          BRANCH_NAME="chore/dep-update-${DATE_TAG}"

          # যদি একই তারিখে আগে থেকে ব্রাঞ্চ থাকে তাহলে ডিলিট করা হচ্ছে
          git push origin --delete "$BRANCH_NAME" 2>/dev/null || true
          git checkout -b "$BRANCH_NAME"

          # পরিবর্তিত ফাইলগুলোর সামারি তৈরি করা হচ্ছে
          CHANGED_FILES=$(git diff --name-only)

          # আপডেট সামারি তৈরি
          SUMMARY="## 📦 Daily Dependency Update — ${DATE_TAG}\n\n"

          if echo "$CHANGED_FILES" | grep -q "pnpm-lock\|package.json"; then
            SUMMARY+="### 🟢 Node.js / pnpm\n"
            SUMMARY+="$(git diff --stat -- '**/package.json' '**/pnpm-lock.yaml' 2>/dev/null || true)\n\n"
          fi

          if echo "$CHANGED_FILES" | grep -q "poetry.lock\|pyproject.toml"; then
            SUMMARY+="### 🐍 Python / Poetry\n"
            SUMMARY+="$(git diff --stat -- '**/pyproject.toml' '**/poetry.lock' 2>/dev/null || true)\n\n"
          fi

          if echo "$CHANGED_FILES" | grep -q "pubspec.lock\|pubspec.yaml"; then
            SUMMARY+="### 🐦 Flutter / Dart\n"
            SUMMARY+="$(git diff --stat -- '**/pubspec.yaml' '**/pubspec.lock' 2>/dev/null || true)\n\n"
          fi

          SUMMARY+="---\n_Auto-generated by SupremeAI Nightly Maintenance_"

          git add -A
          git commit -m "chore(deps): consolidated daily dependency update ${DATE_TAG}"
          git push origin "$BRANCH_NAME"

          # বিদ্যমান PR আছে কিনা চেক — থাকলে আপডেট, না থাকলে নতুন তৈরি
          EXISTING_PR=$(gh pr list --head "$BRANCH_NAME" --json number --jq '.[0].number' 2>/dev/null || true)
          if [ -n "$EXISTING_PR" ] && [ "$EXISTING_PR" != "null" ]; then
            echo "📝 বিদ্যমান PR #${EXISTING_PR} আপডেট করা হচ্ছে..."
            gh pr edit "$EXISTING_PR" --body "$(echo -e "$SUMMARY")"
          else
            echo "🆕 নতুন ড্রাফট PR তৈরি করা হচ্ছে..."
            gh pr create \
              --title "chore(deps): daily dependency update ${DATE_TAG}" \
              --body "$(echo -e "$SUMMARY")" \
              --head "$BRANCH_NAME" \
              --base "${{ github.event.repository.default_branch }}" \
              --draft \
              --label "dependencies"
          fi

          echo "✅ ডিপেন্ডেন্সি আপডেট PR সফলভাবে তৈরি/আপডেট হয়েছে!"
  # ----------------------------------------------------
  # 🖼️ VISUAL REGRESSION TESTS (Heavy UI Audits)
  # ----------------------------------------------------
  visual-regression-test:
    name: 🖼️ Visual Regression Tests
    if: github.event_name == 'schedule' || github.event.inputs.job_to_run == 'all'
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: pnpm/action-setup@v3
        with:
          version: 9.0.0
      - uses: actions/setup-node@v4
        with:
          node-version: ${{ env.NODE_VERSION }}
          cache: 'pnpm'
          cache-dependency-path: '**/pnpm-lock.yaml'
      - name: Install Dependencies
        run: pnpm install --frozen-lockfile
      - name: Install Playwright Browsers
        run: pnpm exec playwright install --with-deps
      - name: Start Frontend Preview Server
        run: |
          cd apps/studio-client && pnpm exec vite preview --port 5173 &
          sleep 5
        env:
          CI: true
      - name: Execute Visual Regression
        run: |
          mkdir -p playwright-report
          pnpm exec playwright test tests/e2e/visual.spec.ts --reporter=html
        env:
          CI: true
      - name: Upload Visual Test Report
        if: always()
        uses: actions/upload-artifact@v4
        with:
          name: visual-regression-report
          path: playwright-report/
          retention-days: 7

  # ----------------------------------------------------
  # ⏱️ LOAD TEST (k6)
  # ----------------------------------------------------
  load-test:
    name: ⏱️ Load Test (k6)
    if: github.event_name == 'schedule' || github.event.inputs.job_to_run == 'all'
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - name: Set up Python
        uses: actions/setup-python@v5
        with:
          python-version: '3.11'
      - name: Start Backend for Testing
        working-directory: backend
        env:
          GCP_SA_KEY: ${{ secrets.GCP_SA_KEY }}
        run: |
          pip install poetry
          poetry config virtualenvs.in-project true
          poetry install --sync --without ml
          
          # Create credentials file for Google Cloud/Firestore
          echo "$GCP_SA_KEY" > $HOME/gcp_key.json
          export GOOGLE_APPLICATION_CREDENTIALS="$HOME/gcp_key.json"
          
          poetry run python main.py &
          sleep 12
      - name: Install k6
        uses: grafana/setup-k6-action@v1
      - name: Run k6 load test
        env:
          SUPREMEAI_URL: "http://127.0.0.1:8000"
        run: |
          echo "Running k6 load test against ${SUPREMEAI_URL}"
          export CI=true
          export SUPREMEAI_URL="${SUPREMEAI_URL}"
          k6 run --out json=load-test-output.json scripts/k6/load_test.js
      - name: Upload k6 results
        uses: actions/upload-artifact@v4
        with:
          name: k6-load-test
          path: load-test-output.json
