+
+from datetime import UTC
+from datetime import datetime
+
+
+def utc_now_iso() -> str:
+    """বর্তমান UTC সময় ISO 8601 ফরম্যাটে রিটার্ন করে।

+    """
+    return datetime.now(UTC)
