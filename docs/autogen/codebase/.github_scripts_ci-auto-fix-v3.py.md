def run_cmd(cmd, cwd=None):
    """Run a shell command and return structured output."""
    if isinstance(cmd, list):
        result = subprocess.run(cmd, capture_output=True, text=True, cwd=cwd)
    else:
        result = subprocess.run(cmd, shell=True, capture_output=True, text=True, cwd=cwd)
    return CmdResult(
        stdout=result.stdout,
        stderr=result.stderr,
        exit_code=result.returncode,
    )

def get_error_severity(error_log: str) -> str:
    """এরর লগের ভিত্তিতে এরর স্যেভারিটি স্কোর নির্ধারণ"""
    critical_patterns = ['security', 'auth', 'vulnerability', 'injection', 'sanitize']
    high_patterns = ['exception', 'failed', 'error', 'crash', 'timeout']
    
    error_lower = error_log.lower()
    if any(p in error_lower for p in critical_patterns):
        return "critical"
    elif any(p in error_lower for p in high_patterns):
        return "high"
    return "low"

def get_tiered_model(error_severity: str) -> Tuple[str, str]:
    """টিয়ার্ড মডেল রাউটিং"""
    if error_severity == "critical":
        return "gemini/gemini-1.5-pro", "critical"
    elif error_severity == "high":
        return "gemini/gemini-2.5-flash", "high"
    else:
        return "gemini/gemini-2.5-flash", "low"

def get_error_hash(error_log: str, file_path: str) -> str:
    combined = f"{file_path}:{error_log[:500]}"
    return hashlib.md5(combined.encode()).hexdigest()[:16]

async def check_cached_fix(error_hash: str) -> Optional[str]:
    if not REDIS_URL or not REDIS_TOKEN:
        return None
    try:
        import httpx
        async with httpx.AsyncClient(timeout=2.0) as client:
            response = await client.get(
                f"{REDIS_URL}/{error_hash}",
                headers={"Authorization": f"Bearer {REDIS_TOKEN}"}
            )
            if response.status_code == 200:
                data = response.json()
                return data.get("fix", None)
    except Exception as e:
        print(f"Redis cache check failed: {e}")
    return None

async def cache_fix(error_hash: str, fix: str) -> None:
    if not REDIS_URL or not REDIS_TOKEN:
        return
    try:
        import httpx
        async with httpx.AsyncClient(timeout=2.0) as client:
            await client.put(
                f"{REDIS_URL}/{error_hash}",
                json={"fix": fix, "cached_at": time.time()},
                headers={
                    "Authorization": f"Bearer {REDIS_TOKEN}",
                    "Content-Type": "application/json"
                }
            )
    except Exception as e:
        print(f"Redis cache set failed: {e}")

def request_admin_approval(filepath: str, diff: str) -> bool:
    if not ADMIN_AUTHORIZED:
        print(f"Critical file {filepath} blocked - admin not authorized")
        return False
    try:
        import urllib.request
        import json
        repo = os.getenv("GITHUB_REPOSITORY", "")
        token = os.getenv("GITHUB_TOKEN", "")
        if repo and token:
            payload = json.dumps({
                "title": f"Critical File Auto-Fix Request: {filepath}",
                "body": f"SupremeAI CI Auto-Fix is requesting changes on a critical file:\n\n**File:** `{filepath}`\n\n**Diff Preview:**\n```\n{diff[:1000]}...\n```\n\nTo approve: comment /approve on this issue",
                "labels": ["auto-fix-pending", "admin-approval"]
            }).encode()
            req = urllib.request.Request(
                f"https://api.github.com/repos/{repo}/issues",
                data=payload,
                headers={
                    "Authorization": f"Bearer {token}",
                    "Accept": "application/vnd.github+json",
                    "Content-Type": "application/json"
                },
                method="POST"
            )
            with urllib.request.urlopen(req) as resp:
                issue_url = json.loads(resp.read().decode()).get("html_url", "")
                print(f"Admin approval issue created: {issue_url}")
                return True
    except Exception as e:
        print(f"Failed to create approval request: {e}")
    return False

def extract_errors(job_type: str, config: dict):
    print(f"Extracting failed logs for job {job_type}...")
    cwd = config.get("cwd")
    result = run_cmd(config["test_cmd"], cwd=cwd)
    stdout = result.stdout + "\n" + result.stderr

    # বাংলা মন্তব্য: এরর লগ থেকে ভুল হওয়া ফাইলের নাম ও পাথ খুঁজে বের করার চেষ্টা করা হচ্ছে
    failing_file = None
    extensions = config["file_exts"]
    
    # Scan for filenames with specified extensions
    matches = []
    for ext in extensions:
        matches.extend(re.findall(rf'([a-zA-Z0-9_\-/]+\{ext})', stdout))

    existing_files = []
    for m in matches:
        p = Path(m)
        if cwd and cwd != ".":
            p_with_cwd = Path(cwd) / m
            if p_with_cwd.exists() and p_with_cwd.is_file():
                existing_files.append(str(p_with_cwd.as_posix()))
                continue
        if p.exists() and p.is_file():
            existing_files.append(str(p.as_posix()))

    # টেস্ট ফাইল বাদ দিয়ে সোর্স ফাইলকে অগ্রাধিকার দেওয়া হচ্ছে
    source_files = [f for f in existing_files if "test" not in f.lower() and "spec" not in f.lower()]
    if source_files:
        failing_file = source_files[0]
    elif existing_files:
        failing_file = existing_files[0]

    if failing_file:
        print(f"Identified potential failing file: {failing_file}")
    else:
        print("Could not identify specific failing file from logs.")

    return stdout, failing_file

def get_fixing_instruction_and_model(file_path: str, error_log: str, comment_char: str) -> Tuple[str, str]:
    error_severity = get_error_severity(error_log)
    preferred_model, tier = get_tiered_model(error_severity)
    default_model = preferred_model if tier != "critical" else "gemini/gemini-2.5-flash"

    default_prompt_template = f"""You are an expert Senior Developer. The CI pipeline just failed.
Analyze the following error log and original file content.

ERROR LOG:
{{error_log}}
{{file_context}}

Provide the fixed complete code for the file that caused the error.
Output ONLY valid code inside a markdown block. Do not include explanations.
Add a comment at the top containing the exact file path like this:
{comment_char} FILE_PATH: {{file_path}}"""

    return default_prompt_template, default_model

def get_ai_fix(error_log: str, file_path: str, comment_char: str) -> str:
    print("Analyzing error and generating fix via SupremeAI...")
    file_context = ""
    if file_path and os.path.exists(file_path):
        try:
            with open(file_path, "r", encoding="utf-8") as f:
                content = f.read()
            file_context = f"\nORIGINAL FILE CONTENT ({file_path}):\n```\n{content}\n```\n"
        except Exception as e:
            print(f"Could not read source file {file_path}: {e}")

    prompt_template, target_model = get_fixing_instruction_and_model(file_path or "example.py", error_log, comment_char)
    try:
        prompt = prompt_template.format(error_log=error_log, file_context=file_context, file_path=file_path or "example.py")
    except Exception:
        prompt = f"{prompt_template}\n\nERROR LOG:\n{error_log}\n{file_context}\nFILE_PATH: {file_path}"

    print(f"Calling model: {target_model} via LiteLLM...")
    try:
        api_key = os.getenv("SUPREMEAI_API_KEY") or os.getenv("GEMINI_API_KEY")
        if api_key:
            os.environ["GEMINI_API_KEY"] = api_key
        response = litellm.completion(
            model=target_model,
            messages=[{"role": "user", "content": prompt}],
            timeout=60.0
        )
        return response.choices[0].message.content
    except Exception as e:
        print(f"Primary model {target_model} call failed: {e}")
        print("Shifting to fallback model gemini/gemini-2.5-flash...")
        response = litellm.completion(
            model="gemini/gemini-2.5-flash",
            messages=[{"role": "user", "content": prompt}],
            timeout=60.0
        )
        return response.choices[0].message.content

# 🧠 STEP 3: APPLY & VALIDATE FIX
# ==========================================
def apply_and_validate_fix(ai_response: str, comment_char: str, config: dict) -> Optional[str]:
    # Extract file path
    path_match = re.search(rf'{re.escape(comment_char)}\s*FILE_PATH:\s*(\S+)', ai_response)
    if not path_match:
        print("ERROR: AI did not provide a valid FILE_PATH comment.")
        sys.exit(1)
        
    file_path = path_match.group(1).strip()
    
    # Clean file path relative to execution
    cwd = config.get("cwd")
    if cwd and cwd != ".":
        if file_path.startswith(cwd + "/"):
            file_path = file_path.replace(cwd + "/", "")

    # Extract code block
    code_match = re.search(r'```[a-zA-Z]*\n(.*?)\n```', ai_response, re.DOTALL)
    if not code_match:
        print("ERROR: AI did not return code inside a markdown block.")
        sys.exit(1)
        
    new_code = code_match.group(1).strip()
    target_write_path = Path(cwd) / file_path if (cwd and cwd != ".") else Path(file_path)

    # Admin check for critical files
    if is_critical_file(str(target_write_path.as_posix())):
        print(f"Critical file detected: {target_write_path}")
        if not AUTOFIX_AUTHORIZED:
            print("Auto-fix not authorized for critical files. Requesting admin approval...")
            diff_result = run_cmd(f"git diff {target_write_path}")
            request_admin_approval(str(target_write_path.as_posix()), diff_result.stdout)
            return None

    # Write file
    target_write_path.parent.mkdir(parents=True, exist_ok=True)
    target_write_path.write_text(new_code, encoding='utf-8')
    print(f"Fix applied to {target_write_path}")

    # Run linter fix command if specified
    if config.get("fix_cmd"):
        print(f"Running lint-fix command: {config['fix_cmd']}")
        run_cmd(config["fix_cmd"], cwd=cwd)

    return str(target_write_path.as_posix())

async def push_fix_to_repo(file_path: str, error_log: str, diff_content: str):
    print("Creating PR for auto-fix...")
    run_cmd('git config --global user.name "SupremeAI Bot"')
    run_cmd('git config --global user.email "bot@supremeai.dev"')
    
    timestamp = int(time.time())
    new_branch = f"auto-fix/{BRANCH}-{timestamp}"
    
    run_cmd(f"git checkout -b {new_branch}")
    run_cmd("git add -A")
    
    commit_msg = f"fix(ai): AI Auto-Fix applied for CI failure\n\nFile: {os.path.basename(file_path)}\nError: {error_log[:200]}..."
    run_cmd(f'git commit -m "{commit_msg}"')
    
    error_hash = get_error_hash(error_log, file_path)
    await cache_fix(error_hash, commit_msg)
    
    run_cmd("git push origin " + new_branch)
    
    # gh CLI to create draft PR
    run_cmd([
        "gh", "pr", "create",
        "--title", f"ci(auto-fix): automated fixes for failed jobs on {BRANCH}",
        "--body", commit_msg + "\n\n**Auto-generated by SupremeAI CI Bot**",
        "--head", new_branch,
        "--base", BRANCH,
        "--draft"
    ])
    print(f"Auto-fix PR successfully requested on branch {new_branch}")

# 🚀 MAIN DISPATCHER ENTRYPOINT
# ==========================================
def main():
    import argparse
    parser = argparse.ArgumentParser(description="Universal Auto-Fixer Engine")
    parser.add_argument("--job", required=True, help="Job/Project type: backend, frontend, mobile")
    parser.add_argument("--mode", default="fix", help="Execution mode (default: fix)")
    args = parser.parse_args()

    # Map job name containing strings to target configs
    api_key = os.getenv("SUPREMEAI_API_KEY") or os.getenv("GEMINI_API_KEY")
    if not api_key:
        print("ERROR: SUPREMEAI_API_KEY or GEMINI_API_KEY is not set.")
        sys.exit(1)

    target_job = None
    for key in JOB_CONFIGS.keys():
        if key in args.job.lower():
            target_job = key
            break

    if not target_job:
        print(f"Unknown job type '{args.job}'. Job configs only support: {list(JOB_CONFIGS.keys())}")
        sys.exit(1)

    print("========================================")
    print(f"   SUPREME-AI AUTO-FIX ENGINE ({target_job.upper()})  ")
    print("========================================")

    check_infinite_loop()
    config = JOB_CONFIGS[target_job]
    error_logs, failing_file = extract_errors(target_job, config)

    if not failing_file:
        print("No failing file could be extracted. Exiting.")
        sys.exit(0)

    # Caching check
    error_hash = get_error_hash(error_logs, failing_file)
    cached_fix = asyncio.run(check_cached_fix(error_hash)) if error_hash else None

    if cached_fix:
        print("Found cached fix for this error - applying without API call!")
        ai_response = f"{config['comment_char']} FILE_PATH: {failing_file}\n\n```{target_job}\n{cached_fix}\n```"
    else:
        ai_response = get_ai_fix(error_logs, failing_file, config["comment_char"])

    fixed_file = apply_and_validate_fix(ai_response, config["comment_char"], config)
    if fixed_file:
        diff_result = run_cmd(f"git diff {fixed_file}")
        asyncio.run(push_fix_to_repo(fixed_file, error_logs[:200], diff_result.stdout))
    else:
        print("Auto-fix blocked or failed approval step.")
        sys.exit(1)

if __name__ == "__main__":
    main()
