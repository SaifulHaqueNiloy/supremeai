```yml
name: 🩹 SupremeAI CI Auto-Fix

on:
  workflow_run:
    workflows: ["Backend CI"] # Or your main CI workflow name
    types:
      - completed

jobs:
  auto-fix:
    runs-on: ubuntu-latest
    # Only run on failures in the main branch, and not on runs triggered by this workflow itself.
    if: >
      github.event.workflow_run.conclusion == 'failure' &&
      github.event.workflow_run.head_branch == 'main' &&
      github.event.workflow_run.actor.login != 'github-actions[bot]'

    steps:
      - name: 📥 Checkout Code
        uses: actions/checkout@v4
        with:
          # We need a token with write permissions to push a new branch
          token: ${{ secrets.GH_PAT_FOR_AUTO_FIX }}

      - name: ⚙️ Set up Python
        uses: actions/setup-python@v4
        with:
          python-version: '3.11'

      - name: 📦 Install Dependencies
        run: |
          python -m pip install --upgrade pip
          pip install ruff # For Python linting/formatting
          npm install -g prettier # For general file formatting

      - name: 🤖 Run Auto-Fix Script
        id: auto_fix_script
        run: |
          # This script will contain the logic to fix common, safe errors.
          echo "Running auto-fixers..."
          FIX_APPLIED="false"

          # 1. Run Prettier to format non-Python files (JSON, YAML, MD, etc.)
          # We ignore errors in case no files are found to format.
          echo "Running Prettier for general formatting..."
          prettier --write "**/*.{json,md,yml,yaml}" --ignore-unknown || true

          # 2. Run ruff to format and fix Python files
          echo "Running ruff for Python formatting..."
          ruff format backend/
          echo "Running ruff for Python linting fixes..."
          ruff --fix backend/

          # Check if any files were changed by the fixer
          if ! git diff --quiet; then
            echo "✅ Fixes were applied by the auto-fix script."
            FIX_APPLIED="true"
          else
            echo "ℹ️ No automatic fixes could be applied."
          fi

          echo "fix_applied=$FIX_APPLIED" >> $GITHUB_OUTPUT

      - name: 🚀 Commit and Create Pull Request
        if: steps.auto_fix_script.outputs.fix_applied == 'true'
        id: create_pr
        run: |
          BRANCH_NAME="auto-fix/ci-${{ github.event.workflow_run.id }}"

          git config --global user.name 'SupremeAI Auto-Fix Bot'
          git config --global user.email 'bot@supremeai.dev'

          echo "Creating new branch: $BRANCH_NAME"
          git checkout -b $BRANCH_NAME

          git add .
          git commit -m "fix(ci): Automatically apply lint fixes

          Triggered by CI run #${{ github.event.workflow_run.id }}"

          echo "Pushing changes to $BRANCH_NAME"
          git push origin $BRANCH_NAME

          PR_TITLE="🩹 CI Auto-Fix for Run #${{ github.event.workflow_run.id }}"
          PR_BODY="This PR was automatically generated by the SupremeAI CI Auto-Fix workflow to address linting issues found in run [#${{ github.event.workflow_run.id }}](https://github.com/${{ github.repository }}/actions/runs/${{ github.event.workflow_run.id }})."

          echo "Creating Pull Request..."
          PR_URL=$(gh pr create --title "$PR_TITLE" \
                               --body "$PR_BODY" \
                               --base main \
                               --head $BRANCH_NAME \
                               --json url -q .url)
          
          echo "Pull Request created: $PR_URL"
          echo "pr_url=$PR_URL" >> $GITHUB_OUTPUT
          echo "pr_title=$PR_TITLE" >> $GITHUB_OUTPUT
        env:
          GH_TOKEN: ${{ secrets.GH_PAT_FOR_AUTO_FIX }}

      - name: 🔔 Notify on Slack
        if: steps.create_pr.outputs.pr_url
        uses: slackapi/slack-github-action@v1.26.0
        with:
          payload: |
            {
              "text": "🤖 SupremeAI Auto-Fix Bot created a new pull request",
              "blocks": [
                {
                  "type": "section",
                  "text": {
                    "type": "mrkdwn",
                    "text": "🩹 *<${{ steps.create_pr.outputs.pr_url }}|${{ steps.create_pr.outputs.pr_title }}>*\nAuto-fix applied for failed CI run. Please review and merge."
                  }
                }
              ]
            }
        env:
          SLACK_WEBHOOK_URL: ${{ secrets.SLACK_WEBHOOK_URL }}
```
